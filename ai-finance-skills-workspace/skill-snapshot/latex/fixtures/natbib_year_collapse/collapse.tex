% Reproduces the natbib year-collapse regression: a citation consolidation
% left the same key twice inside one \citet. Once the work has been cited
% before (so longnamesfirst prints the short author form), natbib merges the
% repeat into the same-author group and renders it as "(2003,?)" — "(2003, 0)"
% under other configurations — with only a log warning.
\documentclass[letterpaper, 12pt]{article}
\usepackage[round,longnamesfirst]{natbib}
\usepackage{setspace}
\usepackage[letterpaper,margin=1in]{geometry}
\bibliographystyle{plainnat}
\begin{document}
\doublespacing
First mention: \citet{kamstra2003winter}.

After the consolidation: \citet{kamstra2003winter,kamstra2003winter,kamstra2015seasonal}.
\bibliography{ref}
\end{document}
