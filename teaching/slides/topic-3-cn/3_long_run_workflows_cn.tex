\begin{frame}{审计重放与确定性}
\begin{itemize}
\setlength{\itemsep}{0.8em}
\item \textbf{审计追溯：}检查当时的证据、操作与审批记录。
\item \textbf{计算复现：}固定输入、代码与配置，再比较结果。
\item \textbf{模型重跑：}衡量决策与工具调用路径的变化。
\item 可重复不等于正确；重放不得重复发出支付或交易指令。
\end{itemize}

% Page 21
\begin{frame}{停止条件：接受、修订或升级处理}
\begin{itemize}
\setlength{\itemsep}{0.8em}
\item \textbf{接受：}检查通过、问题关闭，同一候选策略连续两轮完整审查无问题（QT 规则）。
\item \textbf{修订：}问题可以修复，且仍有预算。
\item \textbf{否决或阻断：}存在致命缺陷或无效证据。
\item \textbf{停止：}预算耗尽、进展停滞或用户叫停。
\end{itemize}
\end{frame}

\begin{frame}{推进条件：什么时候可以进入下一步}
\begin{itemize}
\setlength{\itemsep}{0.8em}
\item \textbf{Generator 关口：}所有已分配专家的报告齐备。
\item \textbf{数值关口：}候选策略实际运行，并通过检查。
\item \textbf{Critic 关口：}审查覆盖完整，前轮遗留问题得到处理。
\item \textbf{终止关口：}证据满足验收或明确停止条件。
\end{itemize}
\end{frame}

% Page 42
\begin{frame}{修复：拒绝不合格的证据\handsup}
\begin{itemize}
\setlength{\itemsep}{0.8em}
\item \textbf{缺少专家报告：}保留在当前处理角色。
\item \textbf{请求已经过期：}拒绝该次提交。
\item \textbf{报告完整且匹配：}检查证据后才能推进。
\item 请在可丢弃的案例中修复并核验路由，不要修改已封存证据。
\end{itemize}
\end{frame}

% Page 43
\begin{frame}{重放：逐项核验证据}
\begin{itemize}
\setlength{\itemsep}{0.8em}
\item \textbf{还原过程：}检查状态、轨迹与保留的产物。
\item \textbf{复算数值：}固定输入，比较数值输出。
\item \textbf{比较行为：}保留不同模型运行的独立记录。
\item 新评测需要新案例；已有案例会接续或复用原有状态。
\end{itemize}
