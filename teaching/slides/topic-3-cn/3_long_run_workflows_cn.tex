\begin{frame}{审计重放与确定性边界}
\begin{itemize}
\setlength{\itemsep}{0.8em}
\item \textbf{审计追溯：}核查当时实际留存的证据文件、执行动作与审批记录。
\item \textbf{计算复现：}锁定输入数据、代码与运行配置，检验数值结果是否一致。
\item \textbf{模型重跑：}衡量再次调用模型时，其决策结论与工具调用路径的波动情况。
\item \textbf{核心边界：}结果可重复并不证明逻辑正确；任何重放都严禁重复发出支付或报单指令。
\end{itemize}
\begin{center}\scriptsize\href{https://arxiv.org/html/2601.15322v4}{来源：DFAH v4（重放的三个维度）}\end{center}

\begin{frame}{终止条件：通过、修订与阻断停止}
\begin{itemize}
\setlength{\itemsep}{0.8em}
\item \textbf{验收通过（Accept）：}门控通过、缺陷清零，且同一候选策略连续两轮完整审查无缺陷（QT 规则）。
\item \textbf{继续修订（Revise）：}存在明确的修复路径，且轮次与时间预算尚未耗尽。
\item \textbf{拒绝或阻断（Reject / Block）：}出现致命缺陷或提交证据不合规时立即停止。
\item \textbf{终止（Stop）：}预算耗尽、反复停滞无进展或用户主动叫停时终止并记录原因。
\end{itemize}
\end{frame}

\begin{frame}{持久状态：落盘保存已核验的进度}
\begin{itemize}
\setlength{\itemsep}{0.8em}
\item \textbf{身份标识：}记录案例、请求、候选策略与提示词版本 ID，直接拒绝过期回复。
\item \textbf{执行进度：}在每次状态转换时落盘保存待处理角色、当前轮次、重试计数与截止时间。
\item \textbf{证据校验：}保存产物文件路径与哈希值，及时发现封存文件是否被改动。
\item \textbf{缺陷台账：}持久化未决缺陷与已核验的修复记录，确保中断后可无缝恢复。
\end{itemize}

% Page 26
\begin{frame}{故障恢复：先核对状态，再决定重试}
\begin{itemize}
\setlength{\itemsep}{0.8em}
\item \textbf{暂时性超时：}先核查任务是否实际完成，再在次数上限内执行重试。
\item \textbf{报告格式违规：}将具体的结构或覆盖缺失错误发回当前角色要求修复。
\item \textbf{封存产物被改：}立即阻断推进（\texttt{BLOCKED}），保留现场并展开排查。
\item \textbf{外部操作未明：}重试前必须先完成状态对账，严防重复发出订单或支付。
\end{itemize}
\end{frame}

\begin{frame}{三个循环：三类截然不同的改动对象}
\begin{itemize}
\setlength{\itemsep}{0.8em}
\item \textbf{工具循环：}在单个角色内部选择下一步工具调用动作。
\item \textbf{研究循环：}围绕当前案例迭代修订候选策略，直至得出终局结论。
\item \textbf{演化循环：}在固定案例集上评估可复用的提示词改动，并决定采纳、拒绝或暂缓。
\item \textbf{关键区分：}调出表面收益更高的策略参数，绝不等于改进了系统的 Harness。
\end{itemize}

\begin{frame}{四道屏障：满足什么条件才允许推进？}
\begin{itemize}
\setlength{\itemsep}{0.8em}
\item \textbf{Generator 汇合屏障：}三位专家报告全部齐备，且请求 ID 与轮次完全匹配。
\item \textbf{数值门控屏障：}候选策略代码实际运行完毕，并通过确定性数值检查。
\item \textbf{Critic 审查屏障：}三份独立审查报告齐备，且逐一处理了前轮遗留缺陷。
\item \textbf{终止判定屏障：}核验后的证据满足验收通过规则，或触发明确的停止状态。
\end{itemize}
\end{frame}

% Page 42
\begin{frame}{故障修复：拦截不合格的提交\handsup}
\begin{itemize}
\setlength{\itemsep}{0.8em}
\item \textbf{缺失专家报告：}汇合屏障直接阻断，案例停留在当前角色等待补齐。
\item \textbf{请求 ID 过期：}直接拒绝错配或过期的提交，严禁推进状态。
\item \textbf{报告齐备且匹配：}校验产物哈希与门控结果后，方可路由至下一环节。
\item \textbf{课堂互动：}在临时案例中修复一次受阻提交并验证路由，全程不得篡改已封存证据。
\end{itemize}
\end{frame}

% Page 43
\begin{frame}{证据重放：三项核验操作对照}
\begin{itemize}
\setlength{\itemsep}{0.8em}
\item \textbf{还原过程：}核查 \texttt{state.json}、路由轨迹与已落盘的角色产物。
\item \textbf{复算数值：}在锁定输入与配置的前提下重新运行数值检查，核对输出是否一致。
\item \textbf{比较模型行为：}跨调用比较模型表现时，分别保存独立的运行记录。
\item \textbf{案例规则：}全新评测必须新建案例；重新打开已有案例只会接续或复用原状态。
\end{itemize}
\end{frame}
