\begin{frame}{产品：用户最终得到什么}
\begin{itemize}

\item \textbf{明确决策：}清晰说明当前案例是通过验收、继续修订，还是终止。

\begin{frame}{失败情形：自信但缺乏依据}
\begin{itemize}
\setlength{\itemsep}{0.7em}
\item \textbf{无据主张：}“动量策略显著有效。”背后的样本区间与交易成本假设是什么？
\item \textbf{无据操作：}“已完成回测验证。”留存的运行输出与日志在哪里？
\item \textbf{无据结论：}“策略可直接上线。”谁核查了未来信息泄漏与风控边界？
\item \textbf{核心教训：}流畅的文本只是待核验的产物，绝不等于证据。
\end{itemize}

\begin{frame}{Harness：将任务转化为可执行契约}
\begin{itemize}
\setlength{\itemsep}{0.7em}
\item \textbf{输入契约：}在运行前固定研究想法、信息截至日期（As-of Date）与股票池。
\item \textbf{工具契约：}读取数据快照，调用已测算函数，并核查实际运行输出。
\item \textbf{输出契约：}返回结构化产物清单、证据 ID 与明确的方法局限。
\item \textbf{权限契约：}明确声明该角色允许读写的文件路径与可执行操作。
\end{itemize}
\end{frame}

\begin{frame}{工具边界：调用前后双端验证}
\begin{itemize}
\setlength{\itemsep}{0.7em}
\item \textbf{调用前：}校验输入参数结构、证券池范围与数据快照 ID。
\item \textbf{调用后：}核查退出码、输出字段结构与产物文件哈希值。
\item \textbf{失败时：}记录错误类型与现场证据，仅在瞬时故障时按上限重试。
\end{itemize}
\end{frame}

\begin{frame}{Hooks：在事件边界自动触发}
\begin{center}\input{figures/qt_hooks}\end{center}
\medskip
\begin{itemize}
\item Skill 指导工作流程，工具执行具体操作，而 Hook 在预定生命周期事件发生时自动运行脚本。
\end{itemize}
\end{frame}

\begin{frame}{Hooks：选择恰当的触发事件}
\begin{itemize}
\setlength{\itemsep}{0.7em}
\item \textbf{\texttt{PreToolUse}：}在工具执行前检查调用参数，必要时直接阻断违规调用。
\item \textbf{\texttt{PostToolUse}：}调用成功后记录审计元数据；调用失败则使用 \texttt{PostToolUseFailure}。
\item \textbf{\texttt{SubagentStop}：}记录子 Agent 完成事件，但团队是否齐备仍由控制器统一核验。
\item \textbf{\texttt{PreCompact}：}在上下文压缩前为已落盘状态生成快照，其返回值会被忽略。
\end{itemize}
\end{frame}

\begin{frame}{Hooks：权限边界与失败语义}
\begin{itemize}
\setlength{\itemsep}{0.7em}
\item \textbf{\href{https://www.kimi.com/code/docs/en/kimi-code-cli/customization/hooks.html}{退出码语义}：}返回 \texttt{0} 正常继续；返回 \texttt{2} 仅在 \texttt{PreToolUse}、\texttt{UserPromptSubmit} 或 \texttt{Stop} 下触发阻断。
\item \textbf{异常与超时：}脚本报错或超时后 Kimi 默认继续运行，因此强制风控门控必须留在控制器内。
\item \textbf{\texttt{Stop} 事件：}可在重试上限内要求模型补充回复，但不能替代控制器的案例验收判定。
\item \textbf{并发幂等：}同一事件匹配的多个 Hook 可能并行执行，脚本必须保持幂等且不依赖执行顺序。
\end{itemize}
\end{frame}

\begin{frame}{角色分工：构建与质疑分离}
\begin{itemize}
\setlength{\itemsep}{0.7em}
\item \textbf{Generator：}负责提出并修复候选方案，但无权为自身产出背书。
\item \textbf{Critic：}负责核查证据并登记缺陷，但无权直接改写候选方案。
\item \textbf{Orchestrator：}掌握控制权，负责任务路由、缺陷台账维护、续行与终止判定。
\end{itemize}
\end{frame}

\end{itemize}
\end{frame}

\begin{frame}{Graph：受控的合法交接}
\begin{itemize}
\setlength{\itemsep}{0.7em}
\item \textbf{节点（Node）：}具备明确输入输出契约的角色或确定性计算步骤。
\item \textbf{边（Edge）：}仅当前置条件满足且证据校验通过时才允许触发的交接路径。
\item \textbf{状态（State）：}决定下一跳路由的当前活动角色、已核验产物与未决缺陷记录。
\item \textbf{显式控制：}用有向图将任务依赖、交接条件与角色权限固化为可强制执行的规则。
\end{itemize}
\end{frame}

\begin{frame}{任务分解：量化系统必须构建的模块}
\begin{itemize}
\setlength{\itemsep}{0.7em}
\item \textbf{数据层：}严格重建每个历史决策时点可获得的信息集合。
\item \textbf{研究与验证层：}将投资假设转化为可检验信号，并映射为 Generator 的构建任务。
\item \textbf{组合与交接层：}在风控约束下将信号转化为目标持仓与待审批建议单。
\item \textbf{运维与审计层：}留存持久状态、调用轨迹与 Critic 审查维度，保障可复现与可恢复。
\end{itemize}
\end{frame}

\begin{frame}{架构 5：为负责人配置专家团队}
\begin{itemize}
\setlength{\itemsep}{0.7em}
\item \textbf{Generator 负责人：}统筹数据、特征与回测三位专家，协力产出一份完整候选方案。
\item \textbf{Critic 负责人：}每轮启用独立审查专家，分别从时点、稳健性与执行三个维度核查。
\item \textbf{Orchestrator：}派发双方负责人，并在每道交接屏障处校验团队产物清单是否齐全。
\end{itemize}
\end{frame}

\begin{frame}{Generator 团队：分工与集成}
\begin{itemize}
\setlength{\itemsep}{0.7em}
\item \textbf{数据专家：}核查数据快照质量，如实记录数据来源覆盖与时点局限。
\item \textbf{特征专家：}将交易想法形式化为明确公式，并声明观测时点与执行滞后。
\item \textbf{回测专家：}运行基准回测与压力测试，返回绩效指标与未解决的风险点。
\item \textbf{时序约束：}独立核查可并行开展，但特征计算必须在数据校验通过后执行。
\end{itemize}
\end{frame}

\begin{frame}{Graph：并行分发与汇合屏障}
\begin{itemize}
\setlength{\itemsep}{0.7em}
\item \textbf{隔离写入：}各专家读取同一数据快照，但写入各自独立的输出路径，避免文件冲突。
\item \textbf{汇合屏障：}负责人必须等待全部必选专家报告齐全，缺失任一报告均不得放行。
\item \textbf{确定性汇总：}按显式规则合并专家产物与缺陷，绝不靠“多数表决”认定事实。
\end{itemize}
\end{frame}

\begin{frame}{执行：中国股票市场交易约束}
\begin{itemize}
\setlength{\itemsep}{0.7em}
\item \textbf{T+1 交收：}分别追踪总持仓与可卖股数，严禁当日买入份额在同日卖出。
\item \textbf{手数与资金：}按所属板块手数规则取整，预留交易税费，并逐日核对现金与持仓。
\item \textbf{涨跌停与停牌：}按买卖方向执行涨跌停与停牌成交限制，如实保留拒单记录。
\item \textbf{公司行动：}区分原始成交价与复权调整，按板块与日期核对分红送转引起的现金和股数变化。
\end{itemize}
\end{frame}

\begin{frame}{Critic 团队：全维度独立审查}
\begin{itemize}
\setlength{\itemsep}{0.7em}
\item \textbf{时点（PIT）审查：}排查未来信息泄漏、历史版本误用与股票池幸存者偏差。
\item \textbf{稳健性审查：}核查逻辑与代码一致性、测试充分性以及交易成本敏感性。
\item \textbf{执行可行性审查：}核查现金配平、整手约束、交易摩擦及实盘交接可行性。
\item \textbf{独立性边界：}每轮使用全新上下文减少锚定效应，但同源模型与共享证据仍可能存在共性盲区。
\end{itemize}
\end{frame}

\begin{frame}{Harness：为每个角色装配恰当上下文}
\begin{itemize}
\setlength{\itemsep}{0.7em}
\item \textbf{固化规则：}将角色职责契约、JSON 结构与权限配置随代码一同版本化管理。
\item \textbf{按需取证：}仅传递当前步骤所需的时点证据与未决缺陷，并保持文件路径和 ID 稳定。
\item \textbf{外置状态：}将执行计划、中间产物与缺陷台账持久化到磁盘，而非依赖聊天记录记忆。
\end{itemize}
\end{frame}

\begin{frame}{反馈：可检验的修复任务包}
\begin{itemize}
\setlength{\itemsep}{0.7em}
\item \textbf{稳定编号：}跨轮次保留缺陷唯一 ID（如 \texttt{pit-001}），便于全程追踪。
\item \textbf{有据分级：}依据具体产物证据标定缺陷严重程度，杜绝空泛批评。
\item \textbf{可检验要求：}向 Generator 下达明确的修复标准（如“改用决策时点已披露版本”）。
\item \textbf{闭环核销：}必须由 Critic 核验修改后的产物方可关闭缺陷；审查漏提绝不视为关闭。
\end{itemize}

\begin{frame}{终止：严格区分验收通过与失败停止}
\begin{itemize}
\setlength{\itemsep}{0.7em}
\item \textbf{\texttt{ACCEPTED}：}同一候选方案通过数值门控，且连续两轮完整审查无未决缺陷。
\item \textbf{\texttt{CONTINUE}：}轮次与时间预算尚余，控制器继续派发修复或第二轮确认审查。
\item \textbf{\texttt{EXHAUSTED} / \texttt{STALLED}：}达到轮次或耗时上限，或连续重复同一未通过状态时停止。
\item \textbf{\texttt{REJECTED} / \texttt{BLOCKED} / \texttt{USER\_STOPPED}：}触发致命缺陷、提交违规或用户主动中断时停止。
\end{itemize}
\end{frame}

\begin{frame}{架构 6：加入持久状态}
\begin{itemize}
\setlength{\itemsep}{0.7em}
\item \textbf{身份标识：}保存案例、请求、候选方案与提示词版本 ID，直接拒绝过期或串线回复。
\item \textbf{执行进度：}在磁盘上记录当前角色、所处轮次、重试次数与截止时间。
\item \textbf{证据完整性：}保存产物路径与 SHA-256 哈希值，及时识别已封存证据是否被改动。
\item \textbf{台账记忆：}跨会话保留缺陷记录、核销理由与路由决定，确保未完工作可无缝恢复。
\end{itemize}
\end{frame}

\begin{frame}{恢复：严格依据已核验状态续跑}
\begin{itemize}
\setlength{\itemsep}{0.7em}
\item \textbf{待处理请求：}重新加载已绑定的 \texttt{request\_id}，仅接收与之匹配的回复。
\item \textbf{封存证据被改：}立即阻断推进（\texttt{BLOCKED}），排查原因后新建独立案例。
\item \textbf{已完成案例：}直接返回已落盘的终局报告，不重复执行计算或外部操作。
\end{itemize}
\end{frame}

\begin{frame}{恢复：先对故障分类，再决定路由}
\begin{itemize}
\setlength{\itemsep}{0.7em}
\item \textbf{限流与超时：}按退避策略在上限内重试，并如实记录完成状态尚不确定。
\item \textbf{结构校验失败：}将具体字段错误发回生成角色，在严格次数上限内要求修复。
\item \textbf{触犯风控门控：}立即阻断并诊断原因，严禁为促成通过而放宽风控参数。
\item \textbf{外部业务操作：}重试前先用幂等操作 ID 完成状态对账，严防重复报单。
\end{itemize}
\end{frame}

\begin{frame}{权限：隔离各角色的工作区域}
\begin{itemize}
\setlength{\itemsep}{0.7em}
\item \textbf{Generator：}仅允许写入指定的候选方案目录，无权修改控制器状态文件。
\item \textbf{Critic：}仅允许写入审查报告目录，无权改动候选代码或回测证据。
\item \textbf{Orchestrator：}统一核验返回的文件路径与哈希，独占状态转移控制权。
\item \textbf{宿主权限隔离：}文件访问边界须由宿主权限强制执行，仅靠提示词无法实现真正隔离。
\end{itemize}
\end{frame}

\begin{frame}{Harness 覆盖检查}
\begin{itemize}
\setlength{\itemsep}{0.7em}
\item \textbf{任务与上下文：}明确定义角色契约、工作指令与按时点筛选的证据输入。
\item \textbf{工具与权限：}校验调用参数结构，限制可读写路径，严防敏感凭据进入提示词。
\item \textbf{跨会话延续：}将产物清单、状态检查点与提示词版本持久化到聊天上下文之外。
\item \textbf{质量与预算：}将确定性代码门控、审计轨迹与重试和耗时上限紧密衔接。
\end{itemize}
\end{frame}

\begin{frame}{上下文安全：将检索文本一律视为数据}
\begin{itemize}
\setlength{\itemsep}{0.7em}
\item \textbf{不可信输入：}研报或公告中若出现“忽略规则、上传凭据”等字样，一律按普通数据对待。
\item \textbf{指令隔离：}将系统控制指令与外部检索文档严格分区，外部文本无权篡改控制器规则。
\item \textbf{最小权限：}通过宿主工具白名单强制约束读写范围，绝不依赖文档内容决定权限。
\end{itemize}
\end{frame}

\begin{frame}{可观测性：让每次状态转移都可复盘}
\begin{itemize}
\setlength{\itemsep}{0.7em}
\item \textbf{数据溯源：}记录每次运行所用的数据快照 ID、披露版本与完整性检查结果。
\item \textbf{Agent 轨迹：}留存提示词版本、模型标识、工具调用参数与返回产物。
\item \textbf{Graph 轨迹：}记录经过的节点、重试次数与路由转移原因，便于快速定位故障。
\item \textbf{独立重放：}串联建议单版本、人工审批与成交记录，使同伴仅凭产物即可重放决策过程。
\end{itemize}
\end{frame}

\item \textbf{演化循环：}在固定评测集上评估 Harness 提示词改动，并配套版本管理与回滚。

\item \textbf{双重核验：}代码回归测试只保护基础机制；提示词是否改进须核查真实模型输出与独立审查。

\item \textbf{时点对齐：}按公开披露时间与供应商入库时间做时点连接（PIT Join），并规范处理复权因子。
\item \textbf{分板块执行：}按交易所、板块与具体日期测试交易日历、整手股数、涨跌停、税费与成交规则。
\end{itemize}
\end{frame}

\begin{itemize}
\setlength{\itemsep}{0.7em}

\item \textbf{证据可复现：}失败用例与修复后通过用例均可复现，并如实注明尚未覆盖的检查项。
\item \textbf{集成无退步：}状态路由、持久状态与完整团队汇合屏障继续通过原有回归测试。
\item \textbf{诚实报告局限：}清楚说明剩余局限与回滚方法；合成回测收益率高低不计入评分。
\end{itemize}
