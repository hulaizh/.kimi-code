\begin{frame}{Why Separate Roles?}
\begin{itemize}
\item \textbf{Generator builds}
\begin{itemize}
\item Produces candidates and tests; cannot grant final acceptance.
\end{itemize}
\item \textbf{Critic challenges}
\begin{itemize}
\item Checks evidence in a fresh context; shared mistakes remain possible.
\end{itemize}
\item \textbf{Orchestrator coordinates}
\begin{itemize}
\item Keeps the mandate fixed; controller code validates state transitions.
\end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Role Contracts}
\begin{itemize}
\item \textbf{Orchestrator}
\begin{itemize}
\item Receives the mandate, state and evidence; returns requests and routing decisions.
\end{itemize}
\item \textbf{Generator}
\begin{itemize}
\item Receives constraints and findings; returns a candidate, tests and evidence.
\end{itemize}
\item \textbf{Critic}
\begin{itemize}
\item Receives candidate evidence and fixed criteria; returns findings and a verdict.
\end{itemize}
\item Each role has defined context, tools and write permissions enforced by the host or controller.
\end{itemize}
\end{frame}

\begin{frame}{Contracts and Boundaries}
\begin{center}
\resizebox{.82\textwidth}{!}{\input{figures/untrusted_document}}
\end{center}
\begin{itemize}
\item \textbf{Contract: what work is required?}
\begin{itemize}
\item Fix inputs, permitted actions, output format and completion criteria; link outputs to evidence.
\end{itemize}
\item \textbf{Boundary: what may the role do?}
\begin{itemize}
\item Treat retrieved text as data. Enforce file, network and spending permissions outside the prompt.
\end{itemize}
\item A paid-data request needs scope, supplier, amount, expiry and approver checks. QT has no payment or broker access.
\end{itemize}
\end{frame}

\begin{frame}{Hooks and Controller}
\begin{center}
\resizebox{.82\textwidth}{!}{\input{figures/hook_concept}}
\end{center}
\begin{itemize}
\item \textbf{Hook: an event callback}
\begin{itemize}
\item Runs when the host emits an event; records metadata or blocks where supported.
\end{itemize}
\item \textbf{Controller: workflow authority}
\begin{itemize}
\item Checks the current state and submitted evidence; chooses advance, retry, wait or stop.
\end{itemize}
\item Example: a hook logs a completed tool call; the controller still rejects a missing specialist report.
\end{itemize}
\end{frame}

\begin{frame}{State and Checkpoints}
\begin{center}
\resizebox{.82\textwidth}{!}{\input{figures/interruption_states}}
\end{center}
\begin{itemize}
\item Identity: case, request, candidate and prompt/version IDs.
\item Progress: pending role, round, retry count and deadline.
\item Evidence: file paths, hashes, open findings and verified repairs.
\item Save verified state at every transition; a conversation's last line does not establish completion.
\end{itemize}
\end{frame}

\begin{frame}{Audit Evidence}
\begin{center}
\resizebox{.94\textwidth}{!}{\input{figures/decision_trace}}
\end{center}
\begin{itemize}
\item Save JSON evidence and Git version history; record who approved which action or version, and when.
\item Reconstruct observable actions and files. Recompute frozen calculations separately from a fresh model run.
\item Recorded traces do not reveal hidden model reasoning.
\end{itemize}
\end{frame}
