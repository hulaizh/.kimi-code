\begin{frame}{Finance: Stakes and Accountability}
\begin{itemize}
\item \textbf{Consequences}
\begin{itemize}
\item Some financial actions are difficult to reverse and can affect customers, capital and institutional obligations.
\item Evidence supporting a conclusion does not itself authorize action.
\end{itemize}
\item \textbf{Ownership}
\begin{itemize}
\item The auditor reconstructs inputs, versions and actual actions.
\item The risk manager checks mandate, limits and who can approve or stop the workflow.
\end{itemize}
\item \textbf{Continuity and change}
\begin{itemize}
\item Operations needs verified progress and reconciliation after interruption.
\item The system owner evaluates changes while preserving existing controls and a way back.
\end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{From Stakes to Controls}
\begin{itemize}
\item \textbf{1. Consequences}
\begin{itemize}
\item Financial decisions need bounded authority and accountable owners.
\end{itemize}
\item \textbf{2. Operational demands}
\begin{itemize}
\item Reconstruct the recorded run; recompute frozen calculations; evaluate fresh model behavior separately.
\item Recover from interruption without repeating uncertain consequential actions.
\end{itemize}
\item \textbf{3. Design response}
\begin{itemize}
\item Use gates, evidence, checkpoints, access boundaries and evaluated versions.
\item These controls also apply to other applications with serious consequences.
\end{itemize}
\end{itemize}
\end{frame}

\begin{itemize}
\item Roles receive different instructions, context, tools and write permissions.
\item State permissions in the prompt and enforce them through the host or controller.
\end{itemize}

\begin{frame}{Inside a Harness}
\begin{itemize}
\item (1) Context and instructions: task, dated evidence and role responsibilities.
\item (2) Tools and permissions: allowed actions, data paths and host-enforced boundaries.
\item (3) State and memory: IDs, hashes, findings and read-only context across transitions.
\item (4) Checks and outputs: schema, evidence, success/failure status and return artifact.
\item The harness surrounds each role; the controller owns workflow state and advancement.
\end{itemize}
\end{frame}

\begin{frame}{Contracts and Boundaries}
\begin{itemize}
\item \textbf{Task and artifact contract}
\begin{itemize}
\item Specify task, fixed inputs, permissions, return format and completion/failure conditions.
\item Link every artifact to its request and supporting evidence.
\end{itemize}
\item \textbf{Permission boundary}
\begin{itemize}
\item Retrieved filings and web pages are data, not authority.
\item A paid-data request still needs purpose, supplier, amount, expiry, identity and approver checks.
\item The classroom QT has no payment or broker access.
\end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Hooks and State Controller}
\begin{itemize}
\item \textbf{Hook}
\begin{itemize}
\item Runs a callback when a host event occurs; records event metadata.
\item May block only when the host supports event blocking.
\end{itemize}
\item \textbf{State controller}
\begin{itemize}
\item Validates submitted artifacts, hashes, findings and numerical gates.
\item Decides whether the run advances, retries, waits or stops.
\end{itemize}
\item A hook observes an event; the controller governs workflow state.
\end{itemize}
\end{frame}

\begin{frame}{State and Checkpoints}
\begin{itemize}
\item (1) Identity: case, request, candidate and prompt/version IDs.
\item (2) Progress: pending role, round, retry count and deadline.
\item (3) Evidence: paths and hashes that detect altered artifacts.
\item (4) Findings: open problems and verified repairs.
\item Save verified state at each transition; a conversation's last line does not show which steps finished.
\end{itemize}
\end{frame}

\begin{frame}{Audit Evidence}
\begin{center}
\resizebox{.94\textwidth}{!}{\input{figures/decision_trace}}
\end{center}
\begin{itemize}
\item Save JSON evidence locally and keep Git version history. For each approval, record the approver, timestamp and authorized action or version.
\item Another analyst can reconstruct observable actions and files; the record does not reveal hidden model reasoning.
\end{itemize}
\end{frame}
